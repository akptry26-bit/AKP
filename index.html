CREATE TABLE IF NOT EXISTS public.finance_ledger ( ledger_date text PRIMARY KEY, payload jsonb NOT NULL, updated_at timestamptz NOT NULL DEFAULT now() ); ALTER TABLE public.finance_ledger ENABLE ROW LEVEL SECURITY; DROP POLICY IF EXISTS "Allow all access" ON public.finance_ledger; DROP POLICY IF EXISTS "Valid ledger rows access" ON public.finance_ledger; CREATE POLICY "Valid ledger rows access" ON public.finance_ledger FOR ALL TO anon, authenticated USING (ledger_date ~ '^\d{4}-\d{2}-\d{2}$') WITH CHECK ( ledger_date ~ '^\d{4}-\d{2}-\d{2}$' AND jsonb_typeof(payload) = 'object' );
